THE CODE

import { useRef } from "react";
import InflatableTulip from "./InflatableTulip";
import sprite from "./assets/tulip-sprite.webp";

export default function App() {
  const word = useRef<HTMLHeadingElement>(null);
  const psi = useRef<HTMLSpanElement>(null);
  const hint = useRef<HTMLParagraphElement>(null);

  // p: 0 = fist (deflated) … 1 = open hand (inflated). Runs every frame,
  // so we update styles directly instead of React state (keeps it smooth).
  const onPressure = (p: number) => {
    const w = word.current;
    if (w) {
      w.style.fontVariationSettings = `"wght" ${200 + p * 700}, "wdth" ${55 + p * 95}`;
      w.style.letterSpacing = `${-0.07 + p * 0.03}em`;
      w.style.transform = `scale(${0.92 + p * 0.08}, ${0.8 + p * 0.2})`;
    }
    if (psi.current) psi.current.textContent = (p * 2.5).toFixed(2);
    if (hint.current) hint.current.textContent = p > 0.5 ? "Squeeze to deflate" : "Open to inflate";
  };

  return (
    <main className="screen">
      <h1 ref={word} className="word">INFLATE</h1>

      <div className="tulip">
        <InflatableTulip sprite={sprite} height="70vh" smoothTime={0.7} onPressureChange={onPressure} />
      </div>

      <p ref={hint} className="hint">Open to inflate</p>
      <div className="gauge"><span ref={psi}>2.50</span> psi</div>
    </main>
  );
}
